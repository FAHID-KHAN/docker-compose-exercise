#### **2. `api-gateway/api_gateway.py`**
```python
from flask import Flask, request
from datetime import datetime

app = Flask(__name__)

# Global state variables
state = "INIT"
run_log = []

@app.route("/state", methods=["GET", "PUT"])
def handle_state():
    global state
    if request.method == "GET":
        return state, 200, {"Content-Type": "text/plain"}

    if request.method == "PUT":
        new_state = request.data.decode("utf-8")
        if new_state in ["INIT", "PAUSED", "RUNNING", "SHUTDOWN"] and new_state != state:
            timestamp = datetime.utcnow().isoformat()
            run_log.append(f"{timestamp}: {state} -> {new_state}")
            state = new_state
            if new_state == "SHUTDOWN":
                return "System shutting down...", 200, {"Content-Type": "text/plain"}
            return f"State changed to {new_state}", 200, {"Content-Type": "text/plain"}
        return "Invalid state or no change detected", 400, {"Content-Type": "text/plain"}

@app.route("/request", methods=["GET"])
def handle_request():
    if state == "RUNNING":
        return "Request processed successfully.", 200, {"Content-Type": "text/plain"}
    elif state == "PAUSED":
        return "System is paused.", 403, {"Content-Type": "text/plain"}
    else:
        return "System is not running.", 400, {"Content-Type": "text/plain"}

@app.route("/run-log", methods=["GET"])
def get_run_log():
    return "\n".join(run_log), 200, {"Content-Type": "text/plain"}

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8197)
```

#### **7. `nginx.conf`**
```nginx
events {}

http {
    upstream service1_backend {
        server service1-1:8199;
        server service1-2:8199;
        server service1-3:8199;
    }

    server {
        listen 80;

        location / {
            auth_basic "Restricted Access";
            auth_basic_user_file /etc/nginx/.htpasswd;
            proxy_pass http://service1_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        location /api/ {
            proxy_pass http://api-gateway:8197/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        location /static/ {
            alias /usr/share/nginx/html/;
            autoindex on;
        }
    }
}
```

#### **10. `tests/test_api.py`**
```python
import requests

BASE_URL = "http://localhost:8197"

def test_get_state():
    response = requests.get(f"{BASE_URL}/state")
    assert response.status_code == 200

def test_put_state():
    response = requests.put(f"{BASE_URL}/state", data="RUNNING")
    assert response.status_code == 200
    assert "State changed to RUNNING" in response.text

def test_run_log():
    response = requests.get(f"{BASE_URL}/run-log")
    assert response.status_code == 200

if __name__ == "__main__":
    test_get_state()
    test_put_state()
    test_run_log()
    print("All tests passed.")
```
